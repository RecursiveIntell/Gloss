    27	    if let Some(error) = val.get("error").and_then(|error| error.as_str()) {
    28	        return Err(GlossError::Provider {
    29	            provider: "ollama".into(),
    30	            source: anyhow::anyhow!("Ollama stream error: {error}"),
    31	        });
    32	    }
    33	
    34	    let done = val.get("done").and_then(|d| d.as_bool()).unwrap_or(false);
    35	    let token = val
    36	        .get("message")
    37	        .and_then(|m| m.get("content"))
    38	        .and_then(|c| c.as_str())
    39	        .unwrap_or("")
    40	        .to_string();
    41	    Ok(ChatToken { token, done })
    42	}
