   211	fn digest_text(text: &str) -> String {
   212	    let mut hasher = Sha256::new();
   213	    hasher.update(text.as_bytes());
   214	    format!("{:x}", hasher.finalize())
   215	}

   247	fn parse_optional_f32(value: Option<String>) -> Option<f32> {
   248	    value
   249	        .and_then(|value| value.trim().parse::<f32>().ok())
   250	        .filter(|value| value.is_finite())
   251	}
   252	
   253	fn parse_optional_i64(value: Option<String>) -> Option<i64> {
   254	    value.and_then(|value| value.trim().parse::<i64>().ok())
   255	}
